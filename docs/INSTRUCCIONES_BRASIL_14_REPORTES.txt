# Cambiar la constante:
DICT_RELATORIOS = [str(n) for n in range(1, 15)]   # 1..14 (antes 1..5)


def clean_name(s):
    """Colapsa saltos de línea y espacios (los nombres de Olinda traen '\\n(a)')."""
    return " ".join(str(s or "").split())


def build_dictionary(anomes):
    """
    Conta -> NomeColuna  y  Conta -> reporte, desde Olinda TipoInstituicao=1
    (prudencial), que SÍ tiene los 14 reportes. No se filtra por institución:
    $top=2000 por reporte captura todas las columnas (se deduplica por Conta).
    """
    nombres, conta_reporte = {}, {}
    for rel in DICT_RELATORIOS:
        url = (
            f"{OLINDA}/IfDataValores(AnoMes={anomes},TipoInstituicao=1,Relatorio='{rel}')"
            f"?$select=Conta,NomeColuna&$top=2000&$format=json"
        )
        try:
            for row in http_json(url).get("value", []):
                c = str(row.get("Conta") or "").strip()
                if c and c not in nombres:        # primer reporte que la trae "gana"
                    nombres[c] = clean_name(row.get("NomeColuna"))
                    conta_reporte[c] = int(rel)
        except Exception as e:
            log.warning("diccionario rel=%s: %s", rel, e)
    log.info("Diccionario: %d cuentas mapeadas en %d reportes", len(nombres), len(DICT_RELATORIOS))
    return nombres, conta_reporte


def load_quarter(conn, dt, files, nombres, conta_reporte):
    fmap = {f["f"].split("/")[-1]: f["f"] for f in files}
    cadastro = portal_file(fmap[f"cadastro{dt}_1009.json"])
    valid = {str(e.get("c0")).strip() for e in cadastro if e.get("c0") is not None}

    inst = [
        (COUNTRY, str(e["c0"]).strip(), str(e.get("c2") or "").strip())
        for e in cadastro if e.get("c0") is not None
    ]
    upsert(conn, "instituciones", ["country", "codigo", "razon_social"],
           ["razon_social"], ["country", "codigo"], inst)

    # Fusionar TODOS los dados{dt}_N.json por entidad
    merged = {}  # e -> {conta: valor}
    for name, path in fmap.items():
        if not (name.startswith(f"dados{dt}_") and name.endswith(".json")):
            continue
        dados = portal_file(path)
        for ent in dados.get("values", []):
            e = str(ent.get("e")).strip()
            if e not in valid:                     # solo universo prudencial + indep.
                continue
            bucket = merged.setdefault(e, {})
            for it in ent.get("v", []):
                conta = str(it.get("i")).strip()
                v = it.get("v")
                try:
                    v = int(round(float(v or 0)))
                except (TypeError, ValueError):
                    v = 0
                bucket[conta] = v

    datos, contas = [], set()
    for e, bucket in merged.items():
        for conta, v in bucket.items():
            datos.append((COUNTRY, str(dt), "p", e, conta, 0, 0, 0, 0, v))
            contas.add(conta)

    plan = [(COUNTRY, c, nombres.get(c, c), conta_reporte.get(c)) for c in contas]
    upsert(conn, "plan_cuentas", ["country", "cuenta", "descripcion", "reporte"],
           ["descripcion", "reporte"], ["country", "cuenta"], plan)
    upsert(conn, "datos_financieros",
           ["country", "periodo", "tipo", "ins_cod", "cuenta",
            "monto_clp", "monto_uf", "monto_tc", "monto_ext", "monto_total"],
           ["monto_total"],
           ["country", "periodo", "tipo", "ins_cod", "cuenta"], datos)
    upsert(conn, "carga_log", ["country", "periodo", "archivos_procesados", "estado"],
           ["archivos_procesados", "estado"], ["country", "periodo"],
           [(COUNTRY, str(dt), len(valid), "ok")])
    log.info("dt=%s: %s entidades, %s filas (%s cuentas distintas)",
             dt, len(valid), len(datos), len(contas))
